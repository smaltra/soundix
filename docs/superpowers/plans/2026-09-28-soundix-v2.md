### Task 1: Wireframe model

**Files:** Create `src/core/layout.ts`, `src/core/layout.test.ts`

**Interfaces (produces):**
```ts
export const UI_KINDS = ['button', 'window', 'label'] as const
export type UiKind = (typeof UI_KINDS)[number]
export const UI_COLORS = ['gold', 'blue', 'green', 'purple', 'red', 'gray'] as const
export type UiColor = (typeof UI_COLORS)[number]
export type UiTrigger = 'press' | 'hover' | 'open' | 'close' | 'select'
export const TRIGGERS: Record<UiKind, UiTrigger[]> // button: press, hover; window: open, close, select; label: []
export const DEFAULT_EVENTS: Record<UiTrigger, string> // press click, hover hover, open open, close close, select select
export type WindowContent = 'list' | 'grid'
export interface UiElement {
  id: string; kind: UiKind; text: string; emoji?: string; color: UiColor
  x: number; y: number; w: number; h: number
  opens?: string                 // button → window id
  content?: WindowContent        // window
  items?: number                 // window, 1..60
  sounds: Partial<Record<UiTrigger, string | null>>  // event ids
}
export function elementId(text: string, kind: UiKind, taken: Iterable<string>): string // toFileName(`${text} ${kind}`) made unique
export function createElement(kind: UiKind, text: string, ui: UiElement[], eventIds: string[]): UiElement
export function place(el: UiElement, box: { x: number; y: number; w: number; h: number }): UiElement // clamp in canvas, min 0.03, snap 0.01
export function renameEventInUi(ui: UiElement[], from: string, to: string): UiElement[]
export function renameElementInUi(ui: UiElement[], from: string, to: string): UiElement[] // id + `opens`
export function sanitizeUi(raw: unknown): UiElement[] | null // null: not a list, or an element with no or unknown kind
export const uiData: (ui: UiElement[]) => object[] // plain JSON order: id, kind, text, emoji, color, x, y, w, h, opens, content, items, sounds
```
Rules: default sizes — button 0.1×0.12, label 0.18×0.07, window 0.5×0.7 centred; new elements at the centre of free space is not needed: new button/label at (0.45, 0.44), window centred. `createElement` sets each trigger to its default event only if that event id is in `eventIds`. `sanitizeUi`: ids via `toFileName` + unique, numbers clamped (`items` 1–60, default 12), unknown colour → gray, `content` default grid, `opens` kept only if it names a window in the same list, `sounds` keep only the element's triggers with `isFileName` values or null.

### Task 3: soundix.json v2

**Files:** Modify `src/core/soundix-json.ts`, `src/core/soundix-json.test.ts`, `src/core/test-fixtures.ts`

**Interfaces:**
```ts
export interface SoundixJson {
  soundix: 2
  events: { id: string; name: string; file: string | null; volume: number; pitch: number; words: string[]; source?: SoundixSource }[]
  ui: object[] // uiData(ui)
}
export function buildSoundixJson(events: SoundEvent[], entries: ExportEntry[], ui: UiElement[]): SoundixJson
export type ImportError = 'not-json' | 'not-soundix' | 'bad-version' | 'bad-events' | 'bad-ui'
export type ImportResult = { ok: true; events: SoundEvent[]; ui: UiElement[] } | { ok: false; error: ImportError }
export function parseSoundixJson(text: string, current: SoundEvent[]): ImportResult
```
Rules: version 1 → events as in v1, `ui: []`; version 2 → events optional (missing → `current`), `ui` must pass `sanitizeUi` else `bad-ui`; events referenced by `ui[].sounds` but missing are created with `createEvent` naming (name = id, no sound). Event validity as after the v1 review (object, string `id`, valid `source`).
